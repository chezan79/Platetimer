
rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    // User profiles are self-readable, but tenant membership is server-controlled.
    // Clients may create/update ordinary profile fields; they may never create,
    // add, remove, or change the company field used by the legacy Service UI.
    match /users/{userId} {
      allow read: if request.auth != null && request.auth.uid == userId;
      allow create: if request.auth != null
                    && request.auth.uid == userId
                    && !('company' in request.resource.data);
      allow update: if request.auth != null
                    && request.auth.uid == userId
                    && !request.resource.data.diff(resource.data).affectedKeys().hasAny(['company']);
      allow delete: if request.auth != null && request.auth.uid == userId;
    }
    
    // Regole per i countdown - solo utenti autenticati
    match /countdowns/{companyId}/{document=**} {
      allow read, write: if request.auth != null;
    }
    
    // Regole generali per documenti pubblici (se necessari)
    match /public/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // Nega tutto il resto
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
